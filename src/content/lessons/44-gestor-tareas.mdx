## Descripción del proyecto

Crearás un **gestor de tareas** (todo list) con estas funcionalidades:

- Agregar tareas con nombre y estado
- Marcar tareas como completadas
- Listar tareas (todas, pendientes o completadas)
- Eliminar tareas
- Persistencia en archivo CSV

## Paso 1: Clase Tarea

<CodeBlock lang="java" title="Modelo de Tarea">
{`import java.time.LocalDate;

public class Tarea {
    private static int contador = 1;
    private int id;
    private String nombre;
    private boolean completada;
    private LocalDate fechaCreacion;

    public Tarea(String nombre) {
        this.id = contador++;
        this.nombre = nombre;
        this.completada = false;
        this.fechaCreacion = LocalDate.now();
    }

    // constructor para cargar desde CSV
    public Tarea(int id, String nombre, boolean completada, LocalDate fecha) {
        this.id = id;
        this.nombre = nombre;
        this.completada = completada;
        this.fechaCreacion = fecha;
        if (id >= contador) contador = id + 1;
    }

    public int getId() { return id; }
    public String getNombre() { return nombre; }
    public boolean isCompletada() { return completada; }
    public void marcarCompletada() { this.completada = true; }
    public LocalDate getFechaCreacion() { return fechaCreacion; }

    @Override
    public String toString() {
        String estado = completada ? "✓" : "○";
        return String.format("%d. [%s] %s (%s)",
            id, estado, nombre, fechaCreacion);
    }

    public String toCSV() {
        return id + "," + nombre + "," + completada + "," + fechaCreacion;
    }
}`}
</CodeBlock>

## Paso 2: Gestor de Tareas (lógica de negocio)

<CodeBlock lang="java" title="Clase Gestor con CRUD">
{`import java.util.*;
import java.util.stream.Collectors;

public class GestorTareas {
    private ArrayList<Tarea> tareas;

    public GestorTareas() {
        this.tareas = new ArrayList<>();
    }

    public void agregar(String nombre) {
        if (nombre == null || nombre.isBlank()) {
            System.out.println("⚠ El nombre no puede estar vacío");
            return;
        }
        tareas.add(new Tarea(nombre));
        System.out.println("✓ Tarea agregada: " + nombre);
    }

    public void marcarCompletada(int id) {
        Tarea tarea = buscarPorId(id);
        if (tarea == null) {
            System.out.println("⚠ Tarea no encontrada: " + id);
            return;
        }
        tarea.marcarCompletada();
        System.out.println("✓ Tarea completada: " + tarea.getNombre());
    }

    public void eliminar(int id) {
        Tarea tarea = buscarPorId(id);
        if (tarea == null) {
            System.out.println("⚠ Tarea no encontrada: " + id);
            return;
        }
        tareas.remove(tarea);
        System.out.println("✓ Tarea eliminada: " + tarea.getNombre());
    }

    public void listar() {
        listarFiltrado(null);
    }

    public void listarPendientes() {
        listarFiltrado(false);
    }

    public void listarCompletadas() {
        listarFiltrado(true);
    }

    private void listarFiltrado(Boolean estado) {
        if (tareas.isEmpty()) {
            System.out.println("📋 No hay tareas registradas");
            return;
        }

        List<Tarea> filtradas = estado == null
            ? tareas
            : tareas.stream()
                .filter(t -> t.isCompletada() == estado)
                .collect(Collectors.toList());

        if (filtradas.isEmpty()) {
            System.out.println("📋 No hay tareas " +
                (estado == Boolean.TRUE ? "completadas" : "pendientes"));
            return;
        }

        String titulo = estado == null ? "TODAS"
            : estado ? "COMPLETADAS" : "PENDIENTES";
        System.out.println("\\n📋 " + titulo + ":");
        filtradas.forEach(t -> System.out.println("  " + t));
    }

    private Tarea buscarPorId(int id) {
        for (Tarea t : tareas) {
            if (t.getId() == id) return t;
        }
        return null;
    }
}`}
</CodeBlock>

## Paso 3: Persistencia en CSV

<CodeBlock lang="java" title="Guardar y cargar tareas">
{`import java.io.*;
import java.time.LocalDate;
import java.util.ArrayList;

public class PersistenciaTareas {
    public static void guardar(ArrayList<Tarea> tareas, String archivo) {
        try (PrintWriter pw = new PrintWriter(new FileWriter(archivo))) {
            pw.println("id,nombre,completada,fecha");
            for (Tarea t : tareas) {
                pw.println(t.toCSV());
            }
        } catch (IOException e) {
            System.out.println("Error al guardar: " + e.getMessage());
        }
    }

    public static ArrayList<Tarea> cargar(String archivo) {
        ArrayList<Tarea> tareas = new ArrayList<>();
        try (BufferedReader br = new BufferedReader(new FileReader(archivo))) {
            String linea = br.readLine(); // saltar encabezado
            while ((linea = br.readLine()) != null) {
                String[] partes = linea.split(",", 4);
                int id = Integer.parseInt(partes[0]);
                boolean comp = Boolean.parseBoolean(partes[2]);
                LocalDate fecha = LocalDate.parse(partes[3]);
                tareas.add(new Tarea(id, partes[1], comp, fecha));
            }
        } catch (FileNotFoundException e) {
            // archivo nuevo — no hay datos previos
        } catch (IOException e) {
            System.out.println("Error al cargar: " + e.getMessage());
        }
        return tareas;
    }
}`}
</CodeBlock>

## Paso 4: Menú principal

<CodeBlock lang="java" title="Interfaz de usuario">
{`import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        GestorTareas gestor = new GestorTareas();
        Scanner sc = new Scanner(System.in);
        int opcion;

        do {
            System.out.println("\\n=== GESTOR DE TAREAS ===");
            System.out.println("1. Agregar tarea");
            System.out.println("2. Marcar completada");
            System.out.println("3. Eliminar tarea");
            System.out.println("4. Listar todas");
            System.out.println("5. Listar pendientes");
            System.out.println("6. Listar completadas");
            System.out.println("7. Salir");
            System.out.print("Opción: ");
            opcion = sc.nextInt();
            sc.nextLine(); // consumir salto de línea

            switch (opcion) {
                case 1 -> {
                    System.out.print("Nombre: ");
                    gestor.agregar(sc.nextLine());
                }
                case 2 -> {
                    System.out.print("ID de tarea: ");
                    gestor.marcarCompletada(sc.nextInt());
                }
                case 3 -> {
                    System.out.print("ID de tarea: ");
                    gestor.eliminar(sc.nextInt());
                }
                case 4 -> gestor.listar();
                case 5 -> gestor.listarPendientes();
                case 6 -> gestor.listarCompletadas();
                case 7 -> System.out.println("¡Hasta luego!");
                default -> System.out.println("Opción inválida");
            }
        } while (opcion != 7);

        sc.close();
    }
}`}
</CodeBlock>

## Ejercicios de extensión

1. **Prioridades**: agrega enum Prioridad {'{'}ALTA, MEDIA, BAJA{'}'}
2. **Fechas**: agrega fecha de vencimiento con resaltado de tareas vencidas
3. **Categorías**: agrupa tareas por categorías (trabajo, personal, estudio)
4. **Buscar**: permite buscar tareas por texto en el nombre
5. **Ordenar**: ordenar por fecha, prioridad o nombre
