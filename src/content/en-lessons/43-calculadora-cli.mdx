      import java.util.ArrayList;
      import java.util.Scanner;

      public class CalculadoraCLI {
          static ArrayList<String> historial = new ArrayList<>();

          public static void main(String[] args) {
              Scanner sc = new Scanner(System.in);
              int opcion;

              do {
                  mostrarMenu();
                  System.out.print("Elige una opción: ");
                  opcion = sc.nextInt();

                  switch (opcion) {
                      case 1 -> realizarOperacion(sc, '+');
                      case 2 -> realizarOperacion(sc, '-');
                      case 3 -> realizarOperacion(sc, '*');
                      case 4 -> realizarOperacion(sc, '/');
                      case 5 -> mostrarHistorial();
                      case 6 -> System.out.println("¡Hasta luego!");
                      default -> System.out.println("Opción inválida");
                  }
              } while (opcion != 6);

              sc.close();
          }

          static void mostrarMenu() {
              System.out.println("\n=== CALCULADORA CLI ===");
              System.out.println("1. Sumar");
              System.out.println("2. Restar");
              System.out.println("3. Multiplicar");
              System.out.println("4. Dividir");
              System.out.println("5. Ver historial");
              System.out.println("6. Salir");
          }

          static void realizarOperacion(Scanner sc, char op) {
              System.out.print("Primer número: ");
              double a = sc.nextDouble();
              System.out.print("Segundo número: ");
              double b = sc.nextDouble();

              try {
                  double resultado = switch (op) {
                      case '+' -> a + b;
                      case '-' -> a - b;
                      case '*' -> a * b;
                      case '/' -> {
                          if (b == 0) throw new ArithmeticException("División por cero");
                          yield a / b;
                      }
                      default -> 0;
                  };

                  String entrada = a + " " + op + " " + b + " = " + resultado;
                  System.out.println("Resultado: " + entrada);
                  historial.add(entrada);
              } catch (ArithmeticException e) {
                  System.out.println("Error: " + e.getMessage());
              }
          }

          static void mostrarHistorial() {
              System.out.println("\n--- Historial ---");
              if (historial.isEmpty()) {
                  System.out.println("No hay operaciones aún");
              } else {
                  for (int i = 0; i < historial.size(); i++) {
                      System.out.printf("%d. %s%n", i + 1, historial.get(i));
                  }
              }
          }
      }

## Project description

You will build an **interactive terminal calculator** with the following features:

- Interactive menu with numeric options
- Addition, subtraction, multiplication, and division
- Error handling (division by zero)
- History of performed operations
- Clean program exit

    System.out.print("Number 1: ");
    double a = sc.nextDouble();
    System.out.print("Number 2: ");
    double b = sc.nextDouble();

    try {
        double resultado = switch (op) {
