## Project Description

You will build a **library management system** that allows you to:

- Add and search books by ISBN, title, or author
- Register users (students, teachers)
- Lend books to users with a return date
- Return books and calculate late fines
- List available and on-loan books

## Step 1: Book Class

<CodeBlock lang="java" title="Book model">
{`public class Libro {
    private String isbn;
    private String titulo;
    private String autor;
    private int anio;
    private boolean disponible;

    public Libro(String isbn, String titulo, String autor, int anio) {
        this.isbn = isbn;
        this.titulo = titulo;
        this.autor = autor;
        this.anio = anio;
        this.disponible = true;
    }

    public String getIsbn() { return isbn; }
    public String getTitulo() { return titulo; }
    public String getAutor() { return autor; }
    public boolean isDisponible() { return disponible; }
    public void setDisponible(boolean d) { this.disponible = d; }

    @Override
    public String toString() {
        String estado = disponible ? "✓ Available" : "✗ On loan";
        return String.format("%s — %s (%d) | %s",
            titulo, autor, anio, estado);
    }
}`}

## Step 2: Loan Class

<CodeBlock lang="java" title="Loan model with dates">
{`import java.time.LocalDate;
import java.time.Period;

public class Prestamo {
    private Libro libro;
    private String usuarioId;
    private LocalDate fechaPrestamo;
    private LocalDate fechaDevolucion;
    private static final double MULTA_DIARIA = 0.50;

    public Libro getLibro() { return libro; }

    public Prestamo(Libro libro, String usuarioId, int diasPrestamo) {
        this.libro = libro;
        this.usuarioId = usuarioId;
        this.fechaPrestamo = LocalDate.now();
        this.fechaDevolucion = fechaPrestamo.plusDays(diasPrestamo);
        libro.setDisponible(false);
    }

    public double devolver() {
        libro.setDisponible(true);
        LocalDate hoy = LocalDate.now();

        if (hoy.isAfter(fechaDevolucion)) {
            int diasRetraso = Period.between(fechaDevolucion, hoy).getDays();
            double multa = diasRetraso * MULTA_DIARIA;
            System.out.printf("⚠ Late return: %d days, fine: $%.2f%n",
                diasRetraso, multa);
            return multa;
        }
        System.out.println("✓ Returned on time");
        return 0;
    }

    @Override
    public String toString() {
        return String.format("%s → %s (due: %s)",
            libro.getTitulo(), usuarioId, fechaDevolucion);
    }
}`}

## Step 3: Catalog (Library)

<CodeBlock lang="java" title="Catalog management with HashMap">
{`import java.util.*;

public class Biblioteca {
    private HashMap<String, Libro> catalogo;
    private ArrayList<Prestamo> prestamosActivos;

    public Biblioteca() {
        catalogo = new HashMap<>();
        prestamosActivos = new ArrayList<>();
    }

    public void agregarLibro(Libro libro) {
        if (catalogo.containsKey(libro.getIsbn())) {
            System.out.println("⚠ ISBN already exists: " + libro.getIsbn());
            return;
        }
        catalogo.put(libro.getIsbn(), libro);
        System.out.println("✓ Book added: " + libro.getTitulo());
    }

    public Libro buscarPorIsbn(String isbn) {
        return catalogo.get(isbn);
    }

    public ArrayList<Libro> buscarPorTitulo(String titulo) {
        ArrayList<Libro> resultado = new ArrayList<>();
        for (Libro l : catalogo.values()) {
            if (l.getTitulo().toLowerCase().contains(titulo.toLowerCase())) {
                resultado.add(l);
            }
        }
        return resultado;
    }

    public void prestar(String isbn, String usuarioId, int dias) {
        Libro libro = catalogo.get(isbn);
        if (libro == null) {
            System.out.println("⚠ Book not found: " + isbn);
            return;
        }
        if (!libro.isDisponible()) {
            System.out.println("⚠ Book already on loan: " + libro.getTitulo());
            return;
        }
        Prestamo p = new Prestamo(libro, usuarioId, dias);
        prestamosActivos.add(p);
        System.out.println("✓ Loan registered: " + p);
    }

    public void devolver(String isbn) {
        Prestamo prestamo = null;
        for (Prestamo p : prestamosActivos) {
            if (p.getLibro().getIsbn().equals(isbn)) {
                prestamo = p;
                break;
            }
        }
        if (prestamo == null) {
            System.out.println("⚠ No active loan for: " + isbn);
            return;
        }
        double multa = prestamo.devolver();
        prestamosActivos.remove(prestamo);
        System.out.printf("Total fine: $%.2f%n", multa);
    }

    public void listarDisponibles() {
        System.out.println("\\n📚 Available books:");
        catalogo.values().stream()
            .filter(Libro::isDisponible)
            .forEach(l -> System.out.println("  " + l));
    }

    public void listarPrestados() {
        System.out.println("\\n📖 Books on loan:");
        for (Prestamo p : prestamosActivos) {
            System.out.println("  " + p);
        }
    }
}`}
</CodeBlock>

{`import java.util.Scanner;

public class MenuBiblioteca {
    public static void main(String[] args) {
        Biblioteca biblio = new Biblioteca();
        Scanner sc = new Scanner(System.in);
        int opcion;

        // Example data
        biblio.agregarLibro(new Libro("978-1", "Java para todos", "Ana García", 2023));
        biblio.agregarLibro(new Libro("978-2", "POO en Java", "Luis Pérez", 2024));
        biblio.agregarLibro(new Libro("978-3", "Estructuras de Datos", "María López", 2023));

        do {
            System.out.println("\\n=== LIBRARY SYSTEM ===");
            System.out.println("1. View available");
            System.out.println("2. View on loan");
            System.out.println("3. Search book");
            System.out.println("4. Lend book");
            System.out.println("5. Return book");
            System.out.println("6. Add book");
            System.out.println("7. Exit");
            System.out.print("Option: ");
            opcion = sc.nextInt();
            sc.nextLine();

            switch (opcion) {
                case 1 -> biblio.listarDisponibles();
                case 2 -> biblio.listarPrestados();
                case 3 -> {
                    System.out.print("Title to search: ");
                    var encontrados = biblio.buscarPorTitulo(sc.nextLine());
                    encontrados.forEach(l -> System.out.println("  " + l));
                }
                case 4 -> {
                    System.out.print("ISBN: ");
                    String isbn = sc.nextLine();
                    System.out.print("User ID: ");
                    String uid = sc.nextLine();
                    System.out.print("Loan days: ");
                    int dias = sc.nextInt();
                    biblio.prestar(isbn, uid, dias);
                }
                case 5 -> {
                    System.out.print("ISBN to return: ");
                    biblio.devolver(sc.nextLine());
                }
                case 6 -> {
                    System.out.print("ISBN: "); String i = sc.nextLine();
                    System.out.print("Title: "); String t = sc.nextLine();
                    System.out.print("Author: "); String a = sc.nextLine();
                    System.out.print("Year: "); int an = sc.nextInt();
                    biblio.agregarLibro(new Libro(i, t, a, an));
                }
                case 7 -> System.out.println("Goodbye!");
                default -> System.out.println("Invalid option");
            }
        } while (opcion != 7);
        sc.close();
    }
}`}
