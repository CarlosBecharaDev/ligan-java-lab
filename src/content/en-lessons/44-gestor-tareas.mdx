## Project description

You will create a **task manager** (todo list) with these features:

- Add tasks with name and status
- Mark tasks as completed
- List tasks (all, pending, or completed)
- Delete tasks
- Persistence in a CSV file

## Step 1: Task class

<CodeBlock lang="java" title="Task model">
{`import java.time.LocalDate;

public class Tarea {
    private static int contador = 1;
    private int id;
    private String nombre;
    private boolean completada;
    private LocalDate fechaCreacion;

    public Tarea(String nombre) {
        this.id = contador++;
        this.nombre = nombre;
        this.completada = false;
        this.fechaCreacion = LocalDate.now();
    }

    // constructor to load from CSV
    public Tarea(int id, String nombre, boolean completada, LocalDate fecha) {
        this.id = id;
        this.nombre = nombre;
        this.completada = completada;
        this.fechaCreacion = fecha;
        if (id >= contador) contador = id + 1;
    }

    public int getId() { return id; }
    public String getNombre() { return nombre; }
    public boolean isCompletada() { return completada; }
    public void marcarCompletada() { this.completada = true; }
    public LocalDate getFechaCreacion() { return fechaCreacion; }

    @Override
    public String toString() {
        String estado = completada ? "✓" : "○";
        return String.format("%d. [%s] %s (%s)",
            id, estado, nombre, fechaCreacion);
    }

    public String toCSV() {
        return id + "," + nombre + "," + completada + "," + fechaCreacion;
    }
}`}
</CodeBlock>

## Step 2: Task Manager (business logic)

<CodeBlock lang="java" title="Manager class with CRUD">
{`import java.util.*;
import java.util.stream.Collectors;

public class GestorTareas {
    private ArrayList<Tarea> tareas;

    public GestorTareas() {
        this.tareas = new ArrayList<>();
    }

    public void agregar(String nombre) {
        if (nombre == null || nombre.isBlank()) {
            System.out.println("⚠ The name cannot be empty");
            return;
        }
        tareas.add(new Tarea(nombre));
        System.out.println("✓ Task added: " + nombre);
    }

    public void marcarCompletada(int id) {
        Tarea tarea = buscarPorId(id);
        if (tarea == null) {
            System.out.println("⚠ Task not found: " + id);
            return;
        }
        tarea.marcarCompletada();
        System.out.println("✓ Task completed: " + tarea.getNombre());
    }

    public void eliminar(int id) {
        Tarea tarea = buscarPorId(id);
        if (tarea == null) {
            System.out.println("⚠ Task not found: " + id);
            return;
        }
        tareas.remove(tarea);
        System.out.println("✓ Task deleted: " + tarea.getNombre());
    }

    public void listar() {
        listarFiltrado(null);
    }

    public void listarPendientes() {
        listarFiltrado(false);
    }

    public void listarCompletadas() {
        listarFiltrado(true);
    }

    private void listarFiltrado(Boolean estado) {
        if (tareas.isEmpty()) {
            System.out.println("📋 No tasks registered");
            return;
        }

        List<Tarea> filtradas = estado == null
            ? tareas
            : tareas.stream()
                .filter(t -> t.isCompletada() == estado)
                .collect(Collectors.toList());

        if (filtradas.isEmpty()) {
            System.out.println("📋 No tasks " +
                (estado == Boolean.TRUE ? "completed" : "pending"));
            return;
        }

        String titulo = estado == null ? "ALL"
            : estado ? "COMPLETED" : "PENDING";
        System.out.println("\\n📋 " + titulo + ":");
        filtradas.forEach(t -> System.out.println("  " + t));
    }

    private Tarea buscarPorId(int id) {
        for (Tarea t : tareas) {
            if (t.getId() == id) return t;
        }
        return null;
    }
}`}
</CodeBlock>

## Step 3: CSV persistence

<CodeBlock lang="java" title="Save and load tasks">
{`import java.io.*;
import java.time.LocalDate;
import java.util.ArrayList;

public class PersistenciaTareas {
    public static void guardar(ArrayList<Tarea> tareas, String archivo) {
        try (PrintWriter pw = new PrintWriter(new FileWriter(archivo))) {
            pw.println("id,nombre,completada,fecha");
            for (Tarea t : tareas) {
                pw.println(t.toCSV());
            }
        } catch (IOException e) {
            System.out.println("Error saving: " + e.getMessage());
        }
    }

    public static ArrayList<Tarea> cargar(String archivo) {
        ArrayList<Tarea> tareas = new ArrayList<>();
        try (BufferedReader br = new BufferedReader(new FileReader(archivo))) {
            String linea = br.readLine(); // skip header
            while ((linea = br.readLine()) != null) {
                String[] partes = linea.split(",", 4);
                int id = Integer.parseInt(partes[0]);
                boolean comp = Boolean.parseBoolean(partes[2]);
                LocalDate fecha = LocalDate.parse(partes[3]);
                tareas.add(new Tarea(id, partes[1], comp, fecha));
            }
        } catch (FileNotFoundException e) {
            // new file — no previous data
        } catch (IOException e) {
            System.out.println("Error loading: " + e.getMessage());
        }
        return tareas;
    }
}`}
</CodeBlock>

## Step 4: Main menu

<CodeBlock lang="java" title="User interface">
{`import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        GestorTareas gestor = new GestorTareas();
        Scanner sc = new Scanner(System.in);
        int opcion;

        do {
            System.out.println("\\n=== TASK MANAGER ===");
            System.out.println("1. Add task");
            System.out.println("2. Mark completed");
            System.out.println("3. Delete task");
            System.out.println("4. List all");
            System.out.println("5. List pending");
            System.out.println("6. List completed");
            System.out.println("7. Exit");
            System.out.print("Option: ");
            opcion = sc.nextInt();
            sc.nextLine(); // consume newline

            switch (opcion) {
                case 1 -> {
                    System.out.print("Name: ");
                    gestor.agregar(sc.nextLine());
                }
                case 2 -> {
                    System.out.print("Task ID: ");
                    gestor.marcarCompletada(sc.nextInt());
                }
                case 3 -> {
                    System.out.print("Task ID: ");
                    gestor.eliminar(sc.nextInt());
                }
                case 4 -> gestor.listar();
                case 5 -> gestor.listarPendientes();
                case 6 -> gestor.listarCompletadas();
                case 7 -> System.out.println("Goodbye!");
                default -> System.out.println("Invalid option");
            }
        } while (opcion != 7);

        sc.close();
    }
}`}
</CodeBlock>
